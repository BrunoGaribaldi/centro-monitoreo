# =====================================================================
# Despliegue del nginx del HOST (no del container)
# Asume servidor con nginx instalado + dominio apuntando al servidor.
#
# Estrategia: hacemos el despliegue en DOS FASES para resolver el clásico
# huevo-y-gallina del SSL:
#   FASE 1 -> Config solo HTTP (sin :443). nginx arranca, certbot puede
#             validar el dominio.
#   FASE 2 -> Certbot --nginx emite el cert y AGREGA solo el bloque :443
#             con los ssl_certificate llenos + convierte el :80 en
#             redirect.
#
# NOTA: por simplicidad de pruebas el dashboard queda PÚBLICO (sin basic
# auth). Para producción hay que meter auth real (OAuth / login en la
# app / al menos restringir por IP).
# =====================================================================

# ---------------------------------------------------------------------
# PASO 0 — Prerrequisitos
# ---------------------------------------------------------------------
# - DNS: panel.dronefieldoperation.cloud apuntando a la IP del server.
#   Verificalo:  dig +short panel.dronefieldoperation.cloud
# - Containers arriba: docker compose ps  -> los dos en "Up"
# - Paquetes:
sudo apt update
sudo apt install -y nginx certbot python3-certbot-nginx

# =====================================================================
# FASE 1 — Sitio HTTP (sin SSL) para que certbot pueda validar
# =====================================================================
sudo nano /etc/nginx/sites-available/drones.conf

# ---------------- pegar EXACTAMENTE esto ----------------
server {
    listen 80;
    listen [::]:80;
    server_name panel.dronefieldoperation.cloud;

    # 1) Dashboard estático -> container dashboard_drones
    location / {
        proxy_pass http://127.0.0.1:18082;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # 2) Señalización WebRTC / WHEP -> MediaMTX
    location /webrtc/ {
        rewrite ^/webrtc/(.*)$ /$1 break;

        proxy_pass http://127.0.0.1:8889;

        # IMPORTANTE: MediaMTX devuelve un header Location: /dronN/whep/<id>
        # tras el POST inicial de WHEP. Como su path interno no tiene el
        # prefijo /webrtc/, el browser sigue a /dronN/whep/<id> (sin
        # prefijo) y los PATCH de trickle ICE caen en location / -> 405.
        # Esta línea reescribe el Location para que vuelva /webrtc/ adelante.
        proxy_redirect ~^/(.*)$ /webrtc/$1;

        # Defensa: si alguna vez se mete basic auth en nginx, no propagamos
        # las credenciales a MediaMTX (le confundirían el authMethod).
        proxy_set_header Authorization     "";

        proxy_http_version 1.1;
        proxy_set_header Upgrade           $http_upgrade;
        proxy_set_header Connection        "upgrade";
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 86400;
        proxy_buffering    off;
    }
}
# ---------------- fin del pegado ----------------

# Activar el sitio y probar
sudo ln -s /etc/nginx/sites-available/drones.conf /etc/nginx/sites-enabled/
# OJO: NO borres /etc/nginx/sites-enabled/default si el server ya hospeda
# otros sitios -- puede tener configuración compartida importante.
sudo nginx -t
sudo systemctl reload nginx

# ---------------------------------------------------------------------
# PASO 2 — Abrir puertos en el firewall (antes de certbot)
# ---------------------------------------------------------------------
sudo ufw allow 80/tcp           # certbot + redirect
sudo ufw allow 443/tcp          # HTTPS dashboard
sudo ufw allow 1935/tcp         # RTMP de FlightHub 2
sudo ufw allow 8189/udp         # WebRTC ICE (paquetes de video)
sudo ufw reload

# En este punto:
#   curl -I http://panel.dronefieldoperation.cloud
# tendría que devolver 200 OK (dashboard servido directo, sin auth).

# =====================================================================
# FASE 2 — Emitir SSL con certbot
# =====================================================================
sudo certbot --nginx -d panel.dronefieldoperation.cloud
# - Cuando pregunte por redirect HTTP->HTTPS, ELEGÍ "2: Redirect".
# - Certbot va a:
#     a) Agregar dentro del MISMO archivo un bloque "server { listen 443 ssl ...}"
#        con todas las locations duplicadas + las líneas ssl_certificate llenas.
#     b) Modificar el bloque :80 para que devuelva 301 a https://...
#
# Después de certbot:
sudo nginx -t
sudo systemctl reload nginx

# =====================================================================
# URLs finales
# =====================================================================
# Dashboard:      https://panel.dronefieldoperation.cloud
# Publicar RTMP:  rtmp://publisher:<MEDIAMTX_PUBLISH_PASS>@panel.dronefieldoperation.cloud:1935/dron1
#                 rtmp://publisher:<MEDIAMTX_PUBLISH_PASS>@panel.dronefieldoperation.cloud:1935/dron2

# =====================================================================
# Renovación automática del certificado
# =====================================================================
# El paquete certbot ya instala un timer systemd que renueva cada 60 días.
# Verificá que esté activo:
systemctl list-timers | grep certbot
# Y simulá una renovación (sin escribir nada):
sudo certbot renew --dry-run
