# =====================================================================
# Despliegue del nginx del HOST (no del container)
# Asume servidor con nginx instalado + dominio apuntando al servidor.
#
# Estrategia: hacemos el despliegue en DOS FASES para resolver el clásico
# huevo-y-gallina del SSL:
#   FASE 1 -> Config solo HTTP (sin :443). nginx arranca, certbot puede
#             validar el dominio.
#   FASE 2 -> Certbot --nginx emite el cert y AGREGA solo el bloque :443
#             con los ssl_certificate llenos + convierte el :80 en
#             redirect. Después editamos para sumar las locations
#             definitivas y el basic auth.
# =====================================================================

# ---------------------------------------------------------------------
# PASO 0 — Prerrequisitos
# ---------------------------------------------------------------------
# - DNS: panel.dronefieldoperation.cloud apuntando a la IP del server.
#   Verificalo:  dig +short panel.dronefieldoperation.cloud
# - Containers arriba: docker compose ps  -> los dos en "Up"
# - Paquetes:
sudo apt update
sudo apt install -y nginx certbot python3-certbot-nginx apache2-utils

# ---------------------------------------------------------------------
# PASO 1 — Crear el archivo de password para basic auth
# ---------------------------------------------------------------------
sudo htpasswd -c /etc/nginx/.htpasswd_drones operador
# Te pide la clave dos veces. Para agregar más usuarios después,
# corré el mismo comando SIN el flag -c (sino sobreescribe el archivo).

# =====================================================================
# FASE 1 — Sitio HTTP (sin SSL) para que certbot pueda validar
# =====================================================================
sudo nano /etc/nginx/sites-available/drones.conf

# ---------------- pegar EXACTAMENTE esto ----------------
server {
    listen 80;
    listen [::]:80;
    server_name panel.dronefieldoperation.cloud;

    # Certbot escribe sus challenges acá; ese path NO requiere auth.
    location /.well-known/acme-challenge/ {
        root /var/www/html;
        auth_basic off;
    }

    auth_basic           "Centro de Monitoreo";
    auth_basic_user_file /etc/nginx/.htpasswd_drones;

    # 1) Dashboard estático -> container dashboard_drones
    location / {
        proxy_pass http://127.0.0.1:18082;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # 2) Señalización WebRTC / WHEP -> MediaMTX
    location /webrtc/ {
        rewrite ^/webrtc/(.*)$ /$1 break;

        proxy_pass http://127.0.0.1:8889;

        proxy_http_version 1.1;
        proxy_set_header Upgrade           $http_upgrade;
        proxy_set_header Connection        "upgrade";
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 86400;
        proxy_buffering    off;
    }
}
# ---------------- fin del pegado ----------------

# Activar el sitio y probar
sudo ln -s /etc/nginx/sites-available/drones.conf /etc/nginx/sites-enabled/
sudo rm -f /etc/nginx/sites-enabled/default     # opcional: sacar el default
sudo nginx -t
sudo systemctl reload nginx

# ---------------------------------------------------------------------
# PASO 2 — Abrir puertos en el firewall (antes de certbot)
# ---------------------------------------------------------------------
sudo ufw allow 80/tcp           # certbot + redirect
sudo ufw allow 443/tcp          # HTTPS dashboard
sudo ufw allow 1935/tcp         # RTMP de FlightHub 2
sudo ufw allow 8189/udp         # WebRTC ICE (paquetes de video)
sudo ufw reload

# En este punto:
#   curl -I http://panel.dronefieldoperation.cloud
# tendría que devolver 401 (basic auth) - eso prueba que llega bien.

# =====================================================================
# FASE 2 — Emitir SSL con certbot
# =====================================================================
sudo certbot --nginx -d panel.dronefieldoperation.cloud
# - Cuando pregunte por redirect HTTP->HTTPS, ELEGÍ "2: Redirect".
# - Certbot va a:
#     a) Agregar dentro del MISMO archivo un bloque "server { listen 443 ssl ...}"
#        con todas las locations duplicadas + las líneas ssl_certificate llenas.
#     b) Modificar el bloque :80 para que devuelva 301 a https://...
#
# Después de certbot:
sudo nginx -t
sudo systemctl reload nginx

# =====================================================================
# PASO 3 — Verificar y ajustar el archivo resultante
# =====================================================================
sudo nano /etc/nginx/sites-available/drones.conf
#
# Confirmá que el bloque :443 tenga TODO esto:
#   - auth_basic + auth_basic_user_file (heredados o explícitos)
#   - location /          -> proxy_pass http://127.0.0.1:18082
#   - location /webrtc/   -> proxy_pass http://127.0.0.1:8889 (con los headers
#                            de WebSocket y proxy_read_timeout 86400)
#   - location /.well-known/acme-challenge/ con "auth_basic off;"
#     (necesario para que la renovación automática de certbot funcione)
#
# Si certbot duplicó bien el bloque, no hace falta editar nada. Si quedó
# alguna location afuera del :443, copiala desde el :80 original.

# Verificá que esté activo:
systemctl list-timers | grep certbot
# Y simulá una renovación (sin escribir nada):
sudo certbot renew --dry-run
