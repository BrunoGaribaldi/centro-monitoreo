# =====================================================================
# Despliegue del nginx del HOST (no del container)
# Asume servidor con nginx instalado + dominio apuntando al servidor.
#
# Estrategia: hacemos el despliegue en DOS FASES para resolver el clásico
# huevo-y-gallina del SSL:
#   FASE 1 -> Config solo HTTP (sin :443). nginx arranca, certbot puede
#             validar el dominio.
#   FASE 2 -> Certbot --nginx emite el cert y AGREGA solo el bloque :443
#             con los ssl_certificate llenos + convierte el :80 en
#             redirect.
# =====================================================================

# ---------------------------------------------------------------------
# PASO 0 — Prerrequisitos
# ---------------------------------------------------------------------
# - DNS: panel.dronefieldoperation.cloud apuntando a la IP del server.
#   Verificalo:  dig +short panel.dronefieldoperation.cloud
# - Containers arriba: docker compose ps  -> los tres en "Up"
# - Paquetes:
sudo apt update
sudo apt install -y nginx certbot python3-certbot-nginx

# =====================================================================
# FASE 1 — Sitio HTTP (sin SSL) para que certbot pueda validar
# =====================================================================
sudo nano /etc/nginx/sites-available/drones.conf

# ---------------- pegar EXACTAMENTE esto ----------------
server {
    listen 80;
    listen [::]:80;
    server_name panel.dronefieldoperation.cloud;

    # ── 1) Dashboard estático -> container dashboard_drones ───────────
    location / {
        proxy_pass http://127.0.0.1:18082;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # ── 2) Auth service (login + cameras API) ─────────────────────────
    location /center-auth/ {
        proxy_pass         http://127.0.0.1:19100;
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_read_timeout 30;
    }

    # ── 3) Location interna para nginx auth_request ───────────────────
    # Solo nginx puede llamar a este endpoint (directiva `internal`).
    # $webrtc_path lo setea location /webrtc/ justo antes del auth_request.
    # Las variables seteadas con `set` en el request padre son accesibles
    # en los subrequests de auth_request.
    location = /center-auth-internal/verify {
        internal;
        proxy_pass              http://127.0.0.1:19100/center-auth/verify;
        proxy_pass_request_body off;
        proxy_set_header        Content-Length  "";
        proxy_set_header        Host            $host;
        proxy_set_header        X-Original-URI  $webrtc_path;
        # Authorization se hereda automáticamente del request original
    }

    # ── 4) Señalización WebRTC / WHEP -> MediaMTX (protegido con JWT) ─
    location /webrtc/ {
        # Capturar la URI original antes de que el rewrite la modifique.
        # $request_uri no se altera por rewrites (a diferencia de $uri).
        set $webrtc_path $request_uri;

        # Verificar JWT antes de proxear a MediaMTX.
        # Devuelve 200 (ok) o 401/403 (denegado).
        auth_request /center-auth-internal/verify;
        error_page 401 = @whep_unauthorized;
        error_page 403 = @whep_unauthorized;

        rewrite ^/webrtc/(.*)$ /$1 break;

        proxy_pass http://127.0.0.1:8889;

        # IMPORTANTE: MediaMTX devuelve Location: /<path>/whep/<id> sin el
        # prefijo /webrtc/. Esta línea lo reescribe para que el browser
        # encuentre el endpoint correcto en el PATCH de trickle ICE.
        proxy_redirect ~^/(.*)$ /webrtc/$1;

        # Limpiar Authorization antes de llegar a MediaMTX.
        # auth_request ya procesó el JWT; MediaMTX no debe verlo.
        proxy_set_header Authorization     "";

        proxy_http_version 1.1;
        proxy_set_header Upgrade           $http_upgrade;
        proxy_set_header Connection        "upgrade";
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 86400;
        proxy_buffering    off;
    }

    # ── 5) Respuesta cuando auth_request deniega acceso ───────────────
    location @whep_unauthorized {
        return 401 '{"error":"unauthorized"}';
        add_header Content-Type "application/json" always;
    }
}
# ---------------- fin del pegado ----------------

# Activar el sitio y probar
sudo ln -s /etc/nginx/sites-available/drones.conf /etc/nginx/sites-enabled/
# OJO: NO borres /etc/nginx/sites-enabled/default si el server ya hospeda
# otros sitios -- puede tener configuración compartida importante.
sudo nginx -t
sudo systemctl reload nginx

# ---------------------------------------------------------------------
# PASO 2 — Abrir puertos en el firewall (antes de certbot)
# ---------------------------------------------------------------------
sudo ufw allow 80/tcp           # certbot + redirect
sudo ufw allow 443/tcp          # HTTPS dashboard
sudo ufw allow 1935/tcp         # RTMP de FlightHub 2
sudo ufw allow 8189/udp         # WebRTC ICE (paquetes de video)
sudo ufw reload

# En este punto:
#   curl -I http://panel.dronefieldoperation.cloud
# tendría que devolver 200 OK (dashboard con pantalla de login).

# =====================================================================
# FASE 2 — Emitir SSL con certbot
# =====================================================================
sudo certbot --nginx -d panel.dronefieldoperation.cloud
# - Cuando pregunte por redirect HTTP->HTTPS, ELEGÍ "2: Redirect".
# - Certbot va a:
#     a) Agregar dentro del MISMO archivo un bloque "server { listen 443 ssl ...}"
#        con todas las locations duplicadas + las líneas ssl_certificate llenas.
#     b) Modificar el bloque :80 para que devuelva 301 a https://...
#
# Después de certbot:
sudo nginx -t
sudo systemctl reload nginx

# =====================================================================
# ACTUALIZACIÓN — si ya tenés nginx+certbot y solo agregás autenticación
# =====================================================================
# Si el servidor ya estaba corriendo (fase 1 y 2 ya hechas), solo hay
# que editar el archivo existente y agregar los bloques nuevos.
#
# El archivo certbot lo habrá duplicado en un bloque :443.
# Hay que agregar los bloques nuevos en AMBOS bloques (el :80 y el :443)
# o solo en el :443 si el :80 ya solo hace redirect.
#
# Los bloques a agregar son los marcados ── 2), 3), 4) y 5) arriba.
# El bloque ── 4) /webrtc/ REEMPLAZA al /webrtc/ que ya existe.
#
# Después de editar:
sudo nginx -t && sudo systemctl reload nginx

# Publicar RTMP:  rtmp://panel.dronefieldoperation.cloud:1935/dock-cam-dock-1

#                 (etc. — sin user/pass porque FlightHub 2 los ignora)

systemctl list-timers | grep certbot

sudo certbot renew --dry-run
