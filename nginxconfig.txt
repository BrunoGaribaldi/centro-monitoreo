# =====================================================================
# Despliegue del nginx del HOST (no del container)
# Centro de Monitoreo · NQN Petrol
#
# Stack:
#   - dashboard estático (container)  → 127.0.0.1:18082
#   - MediaMTX WebRTC signaling       → 127.0.0.1:8889
#   - auth-service (autorización fina) → 127.0.0.1:19100
#   - Authelia (autenticación)        → 127.0.0.1:9091
#
# Estrategia: dos fases (HTTP → SSL con certbot) para resolver el clásico
# huevo-y-gallina del SSL.
# =====================================================================

# ---------------------------------------------------------------------
# PASO 0 — Prerrequisitos
# ---------------------------------------------------------------------
# - DNS: panel.dronefieldoperation.cloud apuntando a la IP del server.
#   Verificalo:  dig +short panel.dronefieldoperation.cloud
# - Containers arriba: docker compose ps  -> los 4 en "Up"
#   (mediamtx, dashboard_drones, auth_service, authelia)
# - Paquetes:
sudo apt update
sudo apt install -y nginx certbot python3-certbot-nginx

# =====================================================================
# FASE 1 — Sitio HTTP (sin SSL) para que certbot pueda validar
# =====================================================================
sudo nano /etc/nginx/sites-available/drones.conf

# ---------------- pegar EXACTAMENTE esto ----------------
server {
    listen 80;
    listen [::]:80;
    server_name panel.dronefieldoperation.cloud;

    # ── 0) Endpoints internos para auth_request ───────────────────────
    # Estos endpoints solo los puede invocar el propio nginx (directiva
    # `internal`). Validan la sesión contra Authelia y la autorización
    # fina por path contra el auth-service.

    # 0a) Authelia — valida cookie de sesión.
    location = /internal/authelia/authz {
        internal;
        proxy_pass http://127.0.0.1:9091/api/authz/auth-request;
        proxy_pass_request_body off;
        proxy_set_header Content-Length        "";

        # Authelia necesita la URL original para evaluar las reglas de
        # access_control (que matchean por domain/resources).
        proxy_set_header X-Original-Method     $request_method;
        proxy_set_header X-Original-URL        $scheme://$host$request_uri;
        proxy_set_header X-Forwarded-Method    $request_method;
        proxy_set_header X-Forwarded-Proto     $scheme;
        proxy_set_header X-Forwarded-Host      $host;
        proxy_set_header X-Forwarded-Uri       $request_uri;
        proxy_set_header X-Forwarded-For       $remote_addr;
    }

    # 0b) auth-service — chequea allowed_paths del user en companies.yml.
    # Recibe el username vía $remote_user (lo setea auth_request_set en
    # cada location pública). $webrtc_path se setea en /webrtc/ antes
    # del auth_request para preservar la URI original.
    location = /internal/cameras/authz {
        internal;
        proxy_pass              http://127.0.0.1:19100/center-auth/authz-path;
        proxy_pass_request_body off;
        proxy_set_header        Content-Length  "";
        proxy_set_header        Host            $host;
        proxy_set_header        Remote-User     $remote_user;
        proxy_set_header        X-Original-URI  $webrtc_path;
    }

    # 0c) WebRTC combinado: el auth-service llama a Authelia internamente
    # y luego valida el path. Nginx solo admite UNA directiva auth_request
    # por location; este endpoint combina las dos validaciones en uno.
    location = /internal/webrtc/authz {
        internal;
        proxy_pass              http://127.0.0.1:19100/center-auth/webrtc-authz;
        proxy_pass_request_body off;
        proxy_set_header        Content-Length       "";
        proxy_set_header        Host                 $host;
        proxy_set_header        Cookie               $http_cookie;
        proxy_set_header        X-Original-Method    $request_method;
        proxy_set_header        X-Original-URL       $scheme://$host$request_uri;
        proxy_set_header        X-Forwarded-Method   $request_method;
        proxy_set_header        X-Forwarded-Proto    $scheme;
        proxy_set_header        X-Forwarded-Host     $host;
        proxy_set_header        X-Forwarded-Uri      $request_uri;
        proxy_set_header        X-Forwarded-For      $remote_addr;
        proxy_set_header        X-Webrtc-Path        $webrtc_path;
    }

    # ── 1) Portal de Authelia ─────────────────────────────────────────
    location /authelia/ {
        proxy_pass         http://127.0.0.1:9091;
        proxy_set_header   Host               $host;
        proxy_set_header   X-Real-IP          $remote_addr;
        proxy_set_header   X-Forwarded-For    $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto  $scheme;
        proxy_set_header   X-Forwarded-Host   $host;
        proxy_set_header   X-Forwarded-Uri    $request_uri;
        proxy_read_timeout 30;
    }

    # ── 2) Dashboard estático (protegido con Authelia) ────────────────
    location / {
        auth_request /internal/authelia/authz;
        error_page 401 = @to_authelia;

        # Capturar headers que devuelve Authelia para tracking opcional
        auth_request_set $remote_user   $upstream_http_remote_user;
        auth_request_set $remote_groups $upstream_http_remote_groups;

        proxy_pass http://127.0.0.1:18082;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # ── 3) Auth-service /center-auth/cameras (protegido con Authelia) ─
    # El auth-service NO valida sesión: lee Remote-User inyectado por
    # nginx (que viene de Authelia tras el auth_request).
    location /center-auth/ {
        auth_request /internal/authelia/authz;
        error_page 401 = @to_authelia_json;

        auth_request_set $remote_user   $upstream_http_remote_user;
        auth_request_set $remote_groups $upstream_http_remote_groups;
        auth_request_set $remote_name   $upstream_http_remote_name;
        auth_request_set $remote_email  $upstream_http_remote_email;

        proxy_pass         http://127.0.0.1:19100;
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_set_header   Remote-User       $remote_user;
        proxy_set_header   Remote-Groups     $remote_groups;
        proxy_set_header   Remote-Name       $remote_name;
        proxy_set_header   Remote-Email      $remote_email;
        proxy_read_timeout 30;
    }

    # ── 4) Señalización WebRTC / WHEP → MediaMTX ──────────────────────
    # auth_request al endpoint combinado (0c): valida sesión Authelia +
    # autorización fina por path en un solo subrequest.
    # NOTA: nginx solo admite UNA auth_request por location — el endpoint
    # 0c del auth-service realiza las dos validaciones internamente.
    location /webrtc/ {
        # Preservar la URI original (rewrites alteran $uri pero no $request_uri).
        set $webrtc_path $request_uri;

        auth_request /internal/webrtc/authz;
        auth_request_set $remote_user $upstream_http_remote_user;
        error_page 401 = @whep_unauthorized;
        error_page 403 = @whep_unauthorized;

        rewrite ^/webrtc/(.*)$ /$1 break;

        proxy_pass http://127.0.0.1:8889;

        # MediaMTX devuelve Location: /<path>/whep/<id> sin el prefijo
        # /webrtc/. Esto lo reescribe para que el browser pegue el PATCH
        # de trickle ICE en la URL correcta.
        proxy_redirect ~^/(.*)$ /webrtc/$1;

        proxy_http_version 1.1;
        proxy_set_header   Upgrade           $http_upgrade;
        proxy_set_header   Connection        "upgrade";
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_read_timeout 86400;
        proxy_buffering    off;
    }

    # ── 5) Handlers de error ──────────────────────────────────────────
    # Para navegación HTML: redirigir al portal con ?rd=<url-actual>.
    location @to_authelia {
        return 302 $scheme://$host/authelia/?rd=$scheme://$host$request_uri;
    }

    # Para fetches XHR (/center-auth/*): devolver 401 JSON.
    # El frontend detecta 401 y hace window.location al portal.
    location @to_authelia_json {
        default_type application/json;
        return 401 '{"error":"unauthorized","login":"/authelia/"}';
    }

    # Para WHEP (no navegable): devolver 401 JSON.
    location @whep_unauthorized {
        default_type application/json;
        return 401 '{"error":"unauthorized"}';
    }
}
# ---------------- fin del pegado ----------------

# Activar el sitio y probar
sudo ln -s /etc/nginx/sites-available/drones.conf /etc/nginx/sites-enabled/
# OJO: NO borres /etc/nginx/sites-enabled/default si el server ya hospeda
# otros sitios.
sudo nginx -t
sudo systemctl reload nginx

# ---------------------------------------------------------------------
# PASO 2 — Abrir puertos en el firewall (antes de certbot)
# ---------------------------------------------------------------------
sudo ufw allow 80/tcp           # certbot + redirect
sudo ufw allow 443/tcp          # HTTPS dashboard
sudo ufw allow 1935/tcp         # RTMP de FlightHub 2
sudo ufw allow 8189/udp         # WebRTC ICE (paquetes de video)
sudo ufw reload

# En este punto:
#   curl -I http://panel.dronefieldoperation.cloud/
# tendría que devolver 302 redirect a /authelia/.

# =====================================================================
# FASE 2 — Emitir SSL con certbot
# =====================================================================
sudo certbot --nginx -d panel.dronefieldoperation.cloud
# - Cuando pregunte por redirect HTTP→HTTPS, elegí "2: Redirect".
# - Certbot duplica el server{} en un bloque :443 con ssl_certificate
#   llenos y modifica el :80 para que solo haga 301.
#
# Después:
sudo nginx -t
sudo systemctl reload nginx

# =====================================================================
# ACTUALIZACIÓN — si ya tenés nginx+certbot corriendo
# =====================================================================
# Hay que copiar TODOS los bloques (locations + handlers @) al bloque :443
# existente, o al único bloque server{} si todavía no corriste certbot.
#
# Después:
sudo nginx -t && sudo systemctl reload nginx

# =====================================================================
# URLs finales
# =====================================================================
# Dashboard:         https://panel.dronefieldoperation.cloud/
# Portal de Authelia: https://panel.dronefieldoperation.cloud/authelia/
# Publicar RTMP:     rtmp://panel.dronefieldoperation.cloud:1935/<path>
#                    (FlightHub 2 ignora user/pass)

# =====================================================================
# Renovación automática del certificado
# =====================================================================
systemctl list-timers | grep certbot
sudo certbot renew --dry-run
